# KOVA STREAK: one-time fix for PCs where KovaaK's lives under C:\Program Files.
# Chrome refuses to open that folder for websites, so this script creates a live
# mirror at %USERPROFILE%\KovaaK-stats and keeps it synced within a second.
# Run from PowerShell:  irm https://rauder999.github.io/kova-streak/mirror-setup.txt | iex
# Undo later: remove KovaaKStatsMirror from HKCU\...\CurrentVersion\Run, kill powershell.exe.

$ErrorActionPreference = 'Stop'

Write-Host ''
Write-Host '=== KOVA STREAK mirror setup ===' -ForegroundColor Magenta

# --- find the stats folder ---
# The Steam root comes from the registry (Steam writes it there on install,
# wherever it lives), Program Files is kept only as a fallback. Then all
# libraries from each root's libraryfolders.vdf.
$statsTail = 'steamapps\common\FPSAimTrainer\FPSAimTrainer\stats'
$steamRoots = @()
foreach ($rk in @('HKCU:\Software\Valve\Steam', 'HKLM:\SOFTWARE\WOW6432Node\Valve\Steam', 'HKLM:\SOFTWARE\Valve\Steam')) {
  try {
    $rp = Get-ItemProperty -Path $rk -ErrorAction Stop
    foreach ($n in @('SteamPath', 'InstallPath')) {
      if ($rp.$n) { $steamRoots += ($rp.$n -replace '/', '\') }
    }
  } catch { }
}
foreach ($pf in @(${env:ProgramFiles(x86)}, $env:ProgramFiles)) {
  if ($pf) { $steamRoots += (Join-Path $pf 'Steam') }
}
$candidates = @()
foreach ($root in ($steamRoots | Select-Object -Unique)) {
  $candidates += (Join-Path $root $statsTail)
  $vdf = Join-Path $root 'steamapps\libraryfolders.vdf'
  if (Test-Path $vdf) {
    foreach ($m in [regex]::Matches((Get-Content $vdf -Raw), '"path"\s+"([^"]+)"')) {
      $lib = $m.Groups[1].Value -replace '\\\\', '\'
      $candidates += (Join-Path $lib $statsTail)
    }
  }
}
$stats = $null
foreach ($c in ($candidates | Select-Object -Unique)) {
  if (Test-Path $c) { $stats = $c; break }
}
if (-not $stats) {
  Write-Host ''
  Write-Host 'Could not find the KovaaK''s stats folder automatically.' -ForegroundColor Yellow
  Write-Host 'Get the path like this:' -ForegroundColor Yellow
  Write-Host '  1. Open Steam, right-click KovaaK''s -> Manage -> Browse local files.'
  Write-Host '  2. In the Explorer window that opens, go into FPSAimTrainer, then into stats.'
  Write-Host '  3. Click the address bar at the top, Ctrl+C the full path.'
  Write-Host '  4. Come back here, RIGHT-CLICK to paste it, press Enter.'
  $stats = Read-Host 'Path'
  $stats = $stats.Trim('"').Trim()
  if (-not (Test-Path $stats)) { Write-Host 'That path does not exist. Aborting, nothing was changed.' -ForegroundColor Red; exit 1 }
}
Write-Host "Found KovaaK's stats: $stats"

# --- if the folder is not under Program Files, no mirror is needed at all ---
$underPF = $false
foreach ($pf in @(${env:ProgramFiles(x86)}, $env:ProgramFiles)) {
  if ($pf -and $stats.StartsWith($pf, [System.StringComparison]::OrdinalIgnoreCase)) { $underPF = $true }
}
if (-not $underPF) {
  Set-Clipboard $stats
  Write-Host ''
  Write-Host 'Done! No mirror needed: Chrome can open this folder directly.' -ForegroundColor Green
  Write-Host 'The folder path is in your clipboard. Go back to the site, press'
  Write-Host '"Choose stats folder", paste (Ctrl+V), press Enter, hit Select Folder:'
  Write-Host "  $stats" -ForegroundColor Cyan
  exit 0
}

# --- install the live mirror ---
$mirror = Join-Path $env:USERPROFILE 'KovaaK-stats'
$ps1 = Join-Path $env:USERPROFILE 'kovaaks-stats-mirror.ps1'
$vbs = Join-Path $env:USERPROFILE 'kovaaks-stats-mirror.vbs'
New-Item -ItemType Directory -Force -Path $mirror | Out-Null

$watcher = @'
# KOVA STREAK live mirror (installed by mirror-setup).
# Chrome refuses to open folders under Program Files for websites,
# so the tracker reads this mirror instead of the real stats folder.
# A FileSystemWatcher wakes on any change, one robocopy delta pass syncs
# everything; latency is about a second. Off switch: remove the
# KovaaKStatsMirror value from HKCU\...\CurrentVersion\Run, kill powershell.

$src = '__SRC__'
$dst = '__DST__'

# exit 99 = another instance is already alive, the watchdog launcher understands this and will not loop restarts
$mutex = New-Object System.Threading.Mutex($false, 'Local\KovaaKStatsMirror')
try { $acquired = $mutex.WaitOne(0) } catch [System.Threading.AbandonedMutexException] { $acquired = $true }
if (-not $acquired) { exit 99 }

$log = Join-Path $env:USERPROFILE 'kovaaks-stats-mirror.log'
"start $(Get-Date -Format s) pid=$PID src=$src" | Set-Content $log

function Sync-Pass {
    robocopy $src $dst /MIR /R:1 /W:1 /NFL /NDL /NJH /NJS /NP | Out-Null
    "pass $(Get-Date -Format s) rc=$LASTEXITCODE" | Add-Content $log
}

Sync-Pass

# Sleep/resume kills the FileSystemWatcher handle (WaitForChanged throws).
# So: any exception = log it, recreate the watcher, sync and keep going.
$fsw = $null
while ($true) {
    try {
        if (-not $fsw) {
            $fsw = New-Object System.IO.FileSystemWatcher $src
            $fsw.IncludeSubdirectories = $false
        }
        $change = $fsw.WaitForChanged([System.IO.WatcherChangeTypes]::All, 3600000)
        "wake $(Get-Date -Format s) timedOut=$($change.TimedOut) name=$($change.Name)" | Add-Content $log
        Start-Sleep -Milliseconds 300
    } catch {
        "watch error $(Get-Date -Format s): $($_.Exception.Message)" | Add-Content $log
        try { if ($fsw) { $fsw.Dispose() } } catch {}
        $fsw = $null
        Start-Sleep -Seconds 5
    }
    Sync-Pass
}
'@
$watcher = $watcher.Replace('__SRC__', $stats).Replace('__DST__', $mirror)
[System.IO.File]::WriteAllText($ps1, $watcher, (New-Object System.Text.UTF8Encoding($false)))

# Watchdog launcher: keeps the watcher alive. Waits for it to exit and restarts
# it after 30 seconds, except on code 99 (another instance is already running, no duplicate needed).
$launcher = "' KOVA STREAK mirror launcher + watchdog (installed by mirror-setup)`r`n" +
  "Set sh = CreateObject(""Wscript.Shell"")`r`n" +
  "Do`r`n" +
  "  rc = sh.Run(""powershell.exe -NoProfile -ExecutionPolicy Bypass -File """"$ps1"""""", 0, True)`r`n" +
  "  If rc = 99 Then Exit Do`r`n" +
  "  WScript.Sleep 30000`r`n" +
  "Loop`r`n"
[System.IO.File]::WriteAllText($vbs, $launcher, (New-Object System.Text.UTF8Encoding($false)))

# restart cleanly if an older copy is running: watchdog first, then watcher,
# otherwise the watchdog would manage to restart the one we just killed
foreach ($pname in @('wscript.exe', 'powershell.exe')) {
  Get-CimInstance Win32_Process -Filter "Name='$pname'" -ErrorAction SilentlyContinue |
    Where-Object { $_.ProcessId -ne $PID -and $_.CommandLine -match 'kovaaks-stats-mirror' } |
    ForEach-Object { Stop-Process -Id $_.ProcessId -Force -ErrorAction SilentlyContinue }
}
Start-Sleep -Milliseconds 500

Set-ItemProperty -Path 'HKCU:\Software\Microsoft\Windows\CurrentVersion\Run' -Name 'KovaaKStatsMirror' -Value ('wscript.exe "' + $vbs + '"')

# Task Scheduler: restart every 15 minutes and at logon. Modern Standby can
# quietly kill background processes overnight, and the Run key only fires at
# logon, which machines that only ever sleep may not see for weeks. Duplicates
# are killed instantly by the mutex, so frequent launches cost nothing.
try {
  $action = New-ScheduledTaskAction -Execute 'wscript.exe' -Argument ('"' + $vbs + '"')
  $t1 = New-ScheduledTaskTrigger -AtLogOn
  $t2 = New-ScheduledTaskTrigger -Once -At (Get-Date).AddMinutes(2) -RepetitionInterval (New-TimeSpan -Minutes 15) -RepetitionDuration (New-TimeSpan -Days 3650)
  $tset = New-ScheduledTaskSettingsSet -AllowStartIfOnBatteries -DontStopIfGoingOnBatteries -ExecutionTimeLimit ([TimeSpan]::Zero) -MultipleInstances IgnoreNew -StartWhenAvailable
  Register-ScheduledTask -TaskName 'KovaaKStatsMirror' -Action $action -Trigger $t1,$t2 -Settings $tset -Force -ErrorAction Stop | Out-Null
} catch {
  schtasks /Create /F /TN KovaaKStatsMirror /SC MINUTE /MO 15 /TR ('wscript.exe \"' + $vbs + '\"') | Out-Null
}

Start-Process wscript.exe -ArgumentList ('"' + $vbs + '"')

Write-Host ''
Write-Host 'Syncing your stats for the first time...'
Start-Sleep -Seconds 3
$count = (Get-ChildItem $mirror -Filter *.csv -ErrorAction SilentlyContinue | Measure-Object).Count
Write-Host "Done! Mirror ready: $count stats files, live-synced, starts with Windows." -ForegroundColor Green
Write-Host ''
Set-Clipboard $mirror
Write-Host 'The folder path is in your clipboard. Go back to the site, press'
Write-Host '"Choose stats folder", paste (Ctrl+V), press Enter, hit Select Folder:'
Write-Host "  $mirror" -ForegroundColor Cyan
Write-Host ''
Write-Host 'If the clipboard gets overwritten, this works in the picker too:'
Write-Host '  %USERPROFILE%\KovaaK-stats' -ForegroundColor Cyan
Write-Host ''
# visual anchor: an Explorer window with the mirror, the path visible in the address bar
Start-Process explorer.exe -ArgumentList ('"' + $mirror + '"')
